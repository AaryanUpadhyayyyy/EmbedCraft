const Organization = require('../models/Organization');

class AdminController {

    /**
     * Create New Client (Organization + Admin User)
     * POST /api/admin/client
     */
    async createClient(req, res) {
        try {
            const { name, adminEmail, password, planLimit, contractEndDate, app_scheme, bundle_id } = req.body;

            // 1. Validation for App Identity
            if (!app_scheme) {
                return res.status(400).json({ error: 'App Scheme is required for Deep Linking (e.g., "myapp://")' });
            }

            // 2. Check Uniqueness (Manual check for clear error message)
            const existingOrg = await Organization.findOne({ app_scheme });
            if (existingOrg) {
                return res.status(409).json({ error: `App Scheme '${app_scheme}' is already in use by another client.` });
            }

            // 3. Create Organization
            const org = await Organization.create({
                name,
                admin_email: adminEmail,
                app_scheme,
                bundle_id: bundle_id || {}, // Default to empty object if not provided
                plan_limit: planLimit || 100000,
                contract_end_date: contractEndDate ? new Date(contractEndDate) : undefined
            });

            // 4. Create Admin User for this Org
            const user = await User.create({
                email: adminEmail,
                password, // Will be hashed by model
                role: 'client_admin',
                organization_id: org._id
            });

            res.status(201).json({
                message: 'Client created successfully',
                organization: org,
                user: { email: user.email, id: user._id }
            });

        } catch (error) {
            // Handle unique index error from MongoDB if race condition occurs
            if (error.code === 11000 && error.keyPattern && error.keyPattern.app_scheme) {
                return res.status(409).json({ error: `App Scheme is already in use.` });
            }
            res.status(500).json({ error: error.message });
        }
    }

    /**
     * Update Contract Details
     * PATCH /api/admin/client/:orgId/contract
     */
    async updateContract(req, res) {
        try {
            const { orgId } = req.params;
            const { contractEndDate, planLimit, status } = req.body;

            // Prepare update object
            const updateData = {};
            if (contractEndDate !== undefined) updateData.contract_end_date = contractEndDate ? new Date(contractEndDate) : undefined;
            if (planLimit !== undefined) updateData.plan_limit = planLimit;

            // Handle Status & Kill Switch Sync
            if (status) {
                updateData.status = status;
                updateData.is_active = status === 'active'; // Sync kill switch
            }

            const org = await Organization.findByIdAndUpdate(
                orgId,
                updateData,
                { new: true }
            );

            res.json({ message: 'Contract updated', organization: org });
        } catch (error) {
            res.status(500).json({ error: error.message });
        }
    }

    /**
     * Get All Clients
     * GET /api/admin/clients

    /**
     * Get Campaign Stats with Time-Series processing
     * GET /api/v1/admin/analytics/campaign/:campaignId
     */
    async getCampaignStats(req, res) {
        try {
            const { campaignId } = req.params;
            const orgId = req.orgId; // Assuming auth middleware sets this
            const EventLog = require('../models/EventLog');
            const Nudge = require('../models/Nudge');

            // 1. Fetch Campaign Details
            const campaign = await Nudge.findOne({ _id: campaignId, organization_id: orgId }).lean();
            if (!campaign) {
                return res.status(404).json({ error: 'Campaign not found' });
            }

            // Use both _id and nudge_id for EventLog matching
            const nudgeIdVariants = [campaignId, campaign.nudge_id].filter(Boolean);
            const goalEventStr = campaign.goal?.event || null;

            const thirtyDaysAgo = new Date();
            thirtyDaysAgo.setDate(thirtyDaysAgo.getDate() - 30);

            const mongoose = require('mongoose');
            // 2. Powerful Faceted Aggregation mapping everything required by the frontend
            const aggregatedStats = await EventLog.aggregate([
                {
                    $match: {
                        nudge_id: { $in: nudgeIdVariants },
                        organization_id: new mongoose.Types.ObjectId(orgId)
                    }
                },
                {
                    $facet: {
                        // Overall Pipeline
                        overall: [
                            {
                                $group: {
                                    _id: null,
                                    impressions: { $sum: { $cond: [{ $regexMatch: { input: "$event_type", regex: /impression|view|shown|dismiss|close/i } }, 1, 0] } },
                                    clicks: { $sum: { $cond: [{ $regexMatch: { input: "$event_type", regex: /click|tap/i } }, 1, 0] } },
                                    conversions: goalEventStr 
                                        ? { $sum: { $cond: [{ $eq: ["$event_type", goalEventStr] }, 1, 0] } }
                                        : { $sum: { $cond: [{ $regexMatch: { input: "$event_type", regex: /conversion|submit/i } }, 1, 0] } },
                                    dismissals: { $sum: { $cond: [{ $regexMatch: { input: "$event_type", regex: /dismiss|close/i } }, 1, 0] } },
                                    uniqueUsers: { $addToSet: "$user_id" }
                                }
                            }
                        ],
                        // Time Series Pipeline (Last 30 Days Trend)
                        trend: [
                            { $match: { timestamp: { $gte: thirtyDaysAgo } } },
                            {
                                $group: {
                                    _id: { $dateToString: { format: "%Y-%m-%d", date: "$timestamp" } },
                                    impressions: { $sum: { $cond: [{ $regexMatch: { input: "$event_type", regex: /impression|view|shown|dismiss|close/i } }, 1, 0] } },
                                    clicks: { $sum: { $cond: [{ $regexMatch: { input: "$event_type", regex: /click|tap/i } }, 1, 0] } },
                                    dismissals: { $sum: { $cond: [{ $regexMatch: { input: "$event_type", regex: /dismiss|close/i } }, 1, 0] } },
                                }
                            },
                            { $sort: { _id: 1 } }
                        ],
                        // Heatmap Pipeline (Hour + Day of week)
                        heatmap: [
                            { $match: { timestamp: { $gte: thirtyDaysAgo } } },
                            {
                                $group: {
                                    _id: {
                                        day: { $dayOfWeek: "$timestamp" }, // 1 (Sun) to 7 (Sat)
                                        hour: { $hour: "$timestamp" }
                                    },
                                    count: { $sum: 1 }
                                }
                            }
                        ],
                        // Dynamic Audience Breakdown Pipeline
                        audience: [
                            { $match: { metadata: { $exists: true, $type: "object" } } },
                            { $project: { kv: { $objectToArray: "$metadata" } } },
                            { $unwind: "$kv" },
                            { 
                                $match: {
                                    "kv.k": { 
                                        $not: { $regex: /nudgeid|campaign|type|auto_rendered|timestamp|time|date|createdat|updatedat|_id|organization/i } 
                                    },
                                    $or: [
                                        { "kv.v": { $type: "string" } },
                                        { "kv.v": { $type: "number" } },
                                        { "kv.v": { $type: "bool" } }
                                    ]
                                } 
                            },
                            {
                                $group: {
                                    _id: { key: "$kv.k", value: "$kv.v" },
                                    count: { $sum: 1 }
                                }
                            },
                            { $sort: { count: -1 } },
                            { $limit: 15 }
                        ]
                    }
                }
            ]);

            const aggregations = aggregatedStats[0];
            const overallStats = aggregations.overall[0] || { impressions: 0, clicks: 0, conversions: 0, dismissals: 0, uniqueUsers: [] };
            const uniqueUserCount = overallStats.uniqueUsers ? overallStats.uniqueUsers.length : 0;

            // Formulate Audience metadata dynamically
            const rawAudience = aggregations.audience || [];
            const audienceBreakdown = rawAudience.map(a => ({
                property: a._id.key,
                value: a._id.value,
                count: a.count
            }));

            // Seed Audience Breakdown with Campaign Config if empty (e.g. pristine new campaign)
            if (audienceBreakdown.length === 0) {

            const filledTrendData = [];
            for (let i = 29; i >= 0; i--) {

                d.setDate(d.getDate() - i);

                for (let hourIndex = 0; hourIndex < 24; hourIndex++) {
                    const found = aggregations.heatmap.find(h => h._id.day === dayIndex && h._id.hour === hourIndex);
                    heatmapArray.push({
                        day: dayName,
                        hour: hourIndex,
                        value: found ? found.count : 0
                    });
                }
            }

            // 5. Build Funnel Data 
            const funnelData = [

                    name: campaign.name,

                    createdAt: campaign.createdAt

                    impressions: campaign.stats?.impressions || overallStats.impressions,

                    conversions: campaign.stats?.conversions || overallStats.conversions,
                    uniqueUserCount: uniqueUserCount,

// ─── Helper: Format bytes ───
const formatSize = (bytes) => {
    if (bytes === 0) return '0 B';
    const units = ['B', 'KB', 'MB', 'GB'];
    const i = Math.floor(Math.log(bytes) / Math.log(1024));
    return (bytes / Math.pow(1024, i)).toFixed(1) + ' ' + units[i];
};

// ─── Helper: Upload buffer to Cloudinary ───
const uploadToCloudinary = (buffer, options = {}) => {
    return new Promise((resolve, reject) => {
        const stream = cloudinary.uploader.upload_stream(
            {
                folder: CLOUDINARY_FOLDER,
                resource_type: options.resource_type || 'auto',
                ...options
            },
            (error, result) => {
                if (result) resolve(result);
                else reject(error);
            }
        );
        streamifier.createReadStream(buffer).pipe(stream);
    });
};

            const isImage = file.mimetype.startsWith('image/');
            const isVideo = file.mimetype.startsWith('video/');
            const type = isImage ? 'image' : isVideo ? 'video' : 'file';

            let uploadResult;

            if (isImage) {
                // Optimize images with Sharp before uploading
                const optimized = await optimizeImage(file.buffer);
                uploadResult = await uploadToCloudinary(optimized, { resource_type: 'image' });
            } else if (isVideo) {
                uploadResult = await uploadToCloudinary(file.buffer, { resource_type: 'video' });
            } else {
                uploadResult = await uploadToCloudinary(file.buffer, { resource_type: 'raw' });
            }

            const asset = await Asset.create({
                name: file.originalname,
                type,
                mimeType: file.mimetype,
                url: uploadResult.secure_url,
                size: formatSize(file.size),
                sizeBytes: file.size,

const User = require('../models/User');
const Organization = require('../models/Organization');
const jwt = require('jsonwebtoken');

const JWT_SECRET = process.env.JWT_SECRET || 'super_secret_key_change_me';

class AuthController {

    /**
     * Login User (Super Admin or Client)
     * POST /api/auth/login
     */
    async login(req, res) {
        try {
            const { email, password } = req.body;

            // 1. Find User
            console.log('Login attempt for:', email);
            const user = await User.findOne({ email }).select('+password').populate('organization_id');
            if (!user) {
                console.log('User not found');
                return res.status(401).json({ error: 'Invalid credentials' });
            }

            // 2. Check Password
            const isMatch = await user.matchPassword(password);
            console.log('Password match:', isMatch);
            if (!isMatch) {
                return res.status(401).json({ error: 'Invalid credentials' });
            }

            // 3. Check Contract Status (If Client)
            if (user.role !== 'super_admin' && user.organization_id) {
                const org = user.organization_id;

                // Check if Org is Active
                if (!org.is_active) {
                    return res.status(403).json({ error: 'Account is inactive. Contact support.' });
                }

                // Check Contract Expiry
                if (org.contract_end_date && new Date() > new Date(org.contract_end_date)) {
                    return res.status(403).json({ error: 'Contract expired. Please renew.' });
                }
            }

            // 4. Generate Token
            const token = jwt.sign(
                { id: user._id, role: user.role, orgId: user.organization_id?._id },
                JWT_SECRET,
                { expiresIn: '24h' }
            );

            // 5. Return Response
            res.json({
                token,
                user: {
                    id: user._id,
                    email: user.email,
                    role: user.role,
                    organization: user.organization_id
                }
            });

        } catch (error) {
            console.error('Login Error:', error);
            res.status(500).json({ error: 'Server Error' });
        }
    }

    /**
     * Register User (Internal / Super Admin only for now)
     * POST /api/auth/register
     */
    async register(req, res) {
        try {
            const { email, password, role, organization_id } = req.body;

            let user = await User.findOne({ email });
            if (user) {
                return res.status(400).json({ error: 'User already exists' });
            }

            user = await User.create({
                email,
                password,
                role,
                organization_id
            });

            res.status(201).json({ message: 'User created successfully' });
        } catch (error) {
            res.status(500).json({ error: error.message });
        }
    }

    /**
     * Change Password
     * POST /api/auth/change-password
     */
    async changePassword(req, res) {
        try {
            const { currentPassword, newPassword } = req.body;

            // Get user with password field included
            const user = await User.findById(req.user.id).select('+password');

            if (!user) {
                return res.status(404).json({ error: 'User not found' });
            }

            // Verify current password
            const isMatch = await user.matchPassword(currentPassword);
            if (!isMatch) {
                return res.status(400).json({ error: 'Incorrect current password' });
            }

            // Update password (pre-save hook will hash it)
            user.password = newPassword;
            await user.save();

            res.json({ message: 'Password updated successfully' });
        } catch (error) {
            console.error('Change Password Error:', error);
            res.status(500).json({ error: 'Failed to update password' });
        }
    }
}

            headers: { 'Accept': 'application/json' }
        };

        const res = await fetchMethod(url, options);
        if (!res.ok) throw new Error(`HTTP Error ${res.status}`);

        // Flexible array parsing
        let arr = [];
        if (Array.isArray(json)) {
            arr = json;

const Nudge = require('../models/Nudge');

const mongoose = require('mongoose');

class GameController {

    /**
     * Engine Rule Evaluator matches dynamic Mobile Properties to Audience Target Matrix
     */
    evaluateAudienceRules(userProps, rules) {
        if (!rules || rules.length === 0) return true; // Empty segment means All Users
        if (!userProps) return false;

        // By default segments use AND logic across rules
        for (const rule of rules) {
            const userVal = String(userProps[rule.field] || '').toLowerCase();
            const targetVal = String(rule.value || '').toLowerCase();
            
            switch (rule.operator) {
                case 'equals':
                    if (userVal !== targetVal) return false;
                    break;
                case 'not_equals':
                    if (userVal === targetVal) return false;
                    break;
                case 'contains':
                    if (!userVal.includes(targetVal)) return false;
                    break;
                case 'not_contains':
                    if (userVal.includes(targetVal)) return false;
                    break;
                default: 
                    // unsupported operator fails safe
                    return false;
            }
        }
        return true;
    }

    /**
     * The True Zero-Trust Spin Initiative
     * POST /v1/game/initiate
     */
    async initiateSpin(req, res) {
        // Start a MongoDB session for Atomicity (prevent double-spins & inventory race conditions)
        const session = await mongoose.startSession();
        session.startTransaction();

        try {
            const { userId, campaignId } = req.body;
            const orgId = req.orgId || req.body.organization_id;
            if (!orgId) {
                return res.status(401).json({ success: false, error: 'Organization context required' });
            }

            if (!userId || !campaignId) {
                await session.abortTransaction();
                session.endSession();
                return res.status(400).json({ success: false, error: 'userId and campaignId required' });
            }

            const query = {
                organization_id: orgId,
                $or: [{ nudge_id: campaignId }, { _id: campaignId }]
            };
            
            // Allow dashboard users (authType = 'jwt') to spin draft/paused campaigns in preview mode
            if (req.authType !== 'jwt') {
                query.status = 'active';
            }

            // 1. Fetch the Game Engine Config
            const nudge = await Nudge.findOne(query).session(session).lean();

            if (!nudge) {
                await session.abortTransaction();
                session.endSession();
                return res.status(404).json({ success: false, error: 'Active Spin The Wheel game not found' });
            }

            // ✅ FIX: Dashboard stores spinTheWheelConfig in multiple possible locations.
            // Support 'previewConfig' from the body if this is a Dashboard Simulation (JWT) with unsaved changes.
            const stwConfig = (req.authType === 'jwt' && req.body.previewConfig) 
                ? req.body.previewConfig 
                : (nudge.config?.spinTheWheelConfig 
                  || nudge.spinTheWheelConfig 
                  || {
                      winningCriteria: nudge.config?.winningCriteria || 'weight',
                      sections: nudge.config?.sections || [],
                      sectionWinLimit: nudge.config?.sectionWinLimit,
                      maxAttempts: nudge.config?.maxAttempts,
                  });

            const sections = stwConfig.sections;

            if (!sections || sections.length === 0) {
                await session.abortTransaction();
                session.endSession();
                return res.status(404).json({ success: false, error: 'No sections configured for this Spin The Wheel game.' });
            }

            // ✅ Preview Mode Detection (moved earlier to guard spin limits too)
            const isPreviewMode = req.authType === 'jwt' && req.body.previewConfig;

            // 2. Security Check: User Attempts Limits (skip for preview — no ledger entries exist)
            if (!isPreviewMode) {
                const userSpins = await UserLedger.find({
                    organization_id: orgId,
                    end_user_id: userId,
                    campaign_id: campaignId
                }).session(session);

                const totalUserSpins = userSpins.length;
                const maxAttemptsAllowed = stwConfig.maxAttempts || 1; // Default to 1 if not set

                if (totalUserSpins >= maxAttemptsAllowed) {
                    await session.abortTransaction();
                    session.endSession();
                    return res.status(403).json({ success: false, error: 'User has exhausted maximum attempts for this game.' });
                }
            }

            // 3. The Probability Engine
            let totalWeight = 0;
            const validSections = [];
            const globalSectionWinLimit = stwConfig.sectionWinLimit;
            const userProperties = req.body.userProperties || {};
            const isAudienceCriteria = stwConfig.winningCriteria === 'audience';

            // Cross-check Inventory & Audience before including in probability mapping
            for (let i = 0; i < sections.length; i++) {
                const sec = sections[i];
                let isAvailable = true;

                // How many times has THIS USER won THIS SLICE?
                const timesWonByUser = userSpins.filter(u => u.metadata && u.metadata.slice_id === sec.id).length;

                // A. Check Per-User Slice Limit
                if (sec.quantity !== undefined && sec.quantity !== null && sec.quantity !== '') {
                    if (timesWonByUser >= Number(sec.quantity)) {
                        isAvailable = false;
                    }
                }

                // A2. Check Global Per-User Section Win Limit
                if (isAvailable && globalSectionWinLimit !== undefined && globalSectionWinLimit !== null) {
                    if (timesWonByUser >= Number(globalSectionWinLimit)) {
                        isAvailable = false;
                    }
                }

                // B. Checking Global Reward Bank Limits (if linked to a Bank Asset)
                if (isAvailable && sec.rewardId) {
                    const rewardBank = await Reward.findById(sec.rewardId).session(session);
                    if (rewardBank && rewardBank.inventory) {
                        if (rewardBank.inventory.total_quantity > 0 && 
                           (rewardBank.inventory.claimed_quantity >= rewardBank.inventory.total_quantity)) {
                            isAvailable = false; // Out of Stock Globally in the Bank!
                        }
                    }
                }

                // C. Evaluate Audience OR Weight
                if (isAvailable) {
                    const sliceWeight = sec.weight !== undefined && sec.weight !== null ? Number(sec.weight) : 1;

                    if (isAudienceCriteria) {
                        // Audience Based: Only add if user matches audience, but preserve custom weight for multiple matches
                        if (sec.audienceId) {
                            const segment = await Segment.findById(sec.audienceId).session(session).lean();
                            if (segment && segment.rules) {
                                const isMatch = this.evaluateAudienceRules(userProperties, segment.rules);
                                if (isMatch && sliceWeight > 0) {
                                    validSections.push({ ...sec, index: i, calculated_weight: sliceWeight });
                                    totalWeight += sliceWeight;
                                }
                            }
                        }
                    } else {
                        // Weight Based (Default 1 ensures compatibility with generic sections)
                        if (sliceWeight > 0) {
                            totalWeight += sliceWeight;
                            validSections.push({ ...sec, index: i, calculated_weight: sliceWeight });
                        }
                    }
                }
            }

            // D. Fallback if Audience produced absolutely NO matches!
            if (isAudienceCriteria && validSections.length === 0) {
                 // Push the lowest weight or "0 weight" items as fallback (like Better Luck Next Time)
                 for (let i = 0; i < sections.length; i++) {
                     if (!sections[i].audienceId && !sections[i].rewardId) {
                         const fallbackWeight = sections[i].weight !== undefined && sections[i].weight !== null ? Number(sections[i].weight) : 1;
                         if (fallbackWeight > 0) {
                             validSections.push({ ...sections[i], index: i, calculated_weight: fallbackWeight });
                             totalWeight += fallbackWeight;
                         }
                     }
                 }
            }

            if (validSections.length === 0 || totalWeight === 0) {
                 await session.abortTransaction();
                 session.endSession();
                 return res.status(400).json({ success: false, error: 'No available rewards or audiences match the user.' });
            }

            // 4. Cryptographic Math Selection
            const randomThreshold = Math.random() * totalWeight; // Evaluated server-side, 100% secure
            let currentSum = 0;
            let winningSection = null;

            for (const sec of validSections) {
                currentSum += sec.calculated_weight;
                if (randomThreshold <= currentSum) {
                    winningSection = sec;
                    break;
                }
            }

            // Fallback just in case floating point math betrayed us
            if (!winningSection) winningSection = validSections[validSections.length - 1];

            // isPreviewMode already declared above at spin limit guard

            // 5. Reserve & Deduct the Inventory Atomically (Skip for previews)
            if (winningSection.rewardId && !isPreviewMode) {
                 // Decrement Global Database Vault ONLY IF it's a real Reward Object from the Bank!
                 await Reward.findByIdAndUpdate(
                        winningSection.rewardId,
                        { $inc: { 'inventory.claimed_quantity': 1 } },
                        { session }
                 );
            }

            // 6. Create the User Ledger Entry (Skip for previews — no fraud trail needed for simulation)
            let ledgerEntry = null;
            if (!isPreviewMode) {
                ledgerEntry = await UserLedger.create([{
                    organization_id: orgId,
                    end_user_id: userId,
                    campaign_id: campaignId,
                    reward_id: winningSection.rewardId || null,
                    claim_source: 'spin_the_wheel',
                    metadata: {
                        slice_id: winningSection.id,
                        slice_name: winningSection.name
                    },
                    status: 'unlocked' // Unlocked, waiting for user to click "Claim" 
                }], { session });
            }

            // 7. Success! Commit everything
            await session.commitTransaction();
            session.endSession();

            // 8. Output Visual Results to Client SDK
            // We ONLY tell the SDK the winning INDEX. The SDK will rotate (index * 360/total_slices)
            // CRITICAL: Include isWin so SDK can show congrats/betterLuck screens
            // (rewardId is stripped from sections by nudgeController sanitization)
            const isWin = !!(winningSection.rewardId && winningSection.rewardId !== 'no_reward');
            return res.status(200).json({
                success: true,
                winningIndex: winningSection.index, // The exact slice array index
                winningId: winningSection.id,
                isWin, // Server-authoritative win determination (SDK cannot check rewardId — it's sanitized)
                rewardName: winningSection.name, // Safe to expose the name (not the rewardId)
                message: `Landed on ${winningSection.name}`,
                ledgerId: ledgerEntry ? ledgerEntry[0]._id : null, // Null for preview mode
                preview: isPreviewMode || false
            });

        } catch (error) {
            await session.abortTransaction();
            session.endSession();
            console.error('[Game Engine] Initiation Failed:', error);
            return res.status(500).json({ success: false, error: 'Internal Game Engine Error' });
        }
    }

const Nudge = require('../models/Nudge');

const cacheService = require('../services/cacheService');

class NudgeController {

    /**
     * High-scale fetch endpoint
     * GET /api/v1/nudge/fetch?userId=...&screenName=...&platform=...
     */
    async fetchNudge(req, res) {
        try {
            const { userId, screenName, platform } = req.query;
            const orgId = req.orgId; // From Middleware

            if (!screenName) {
                return res.status(400).json({ error: 'screenName is required' });
            }

            // Require Models (Safe Scope)
            const EndUser = require('../models/EndUser');
            const EventLog = require('../models/EventLog');
            const Nudge = require('../models/Nudge');

            // 1. Try Cache First (Cache Key now includes OrgID to prevent mixup)
            const cacheKey = `${orgId}:${screenName}`;
            let nudges = await cacheService.getNudges(cacheKey);

            // 1.5 FAIL-SAFE: Filter Cached Results by Status
            // Even if cache is hit, ensure we ONLY return active campaigns.
            // This fixes the issue where a campaign is deactivated but still lingers in cache.
            if (nudges && Array.isArray(nudges)) {
                nudges = nudges.filter(n => n.status === 'active');
                if (nudges.length === 0) {
                    nudges = null; // Force DB re-fetch if cache only had inactive items
                }
            }

            // 2. If Cache Miss, Fetch from DB
            if (!nudges) {
                try {
                    nudges = await Nudge.find({
                        organization_id: orgId,
                        status: 'active',
                        $or: [
                            { 'display_rules.pages': { $in: [screenName] } }, // Explicit match in array
                            { 'display_rules.pages': { $size: 0 } },          // Empty array = All pages
                            { 'display_rules.pages': { $exists: false } },    // Field missing = All pages (legacy)
                            { trigger_screen: 'all' },                        // Legacy "all"
                            { trigger_screen: screenName }                    // Legacy exact match
                        ]
                    }).sort({ priority: -1 }).lean();

                    // 3. Update Cache (Async)
                    if (nudges) {
                        cacheService.setNudges(cacheKey, nudges);
                    }
                } catch (dbError) {
                    console.error('DB Error:', dbError.message);
                    // Silent fail: If DB is down, nudges remains null/empty
                    nudges = [];
                }
            }

            // 2.5 Fetch User Profile for Targeting
            const userProfile = await EndUser.findOne({ organization_id: orgId, user_id: userId }).lean();
            const userProperties = userProfile ? { ...userProfile.properties, ...userProfile } : {};

            // ========== 🚀 OPTIMIZED QUERY BATCHING - 99% Faster ==========
            // Instead of 250-500 queries, we do 3 batched queries!

            let matchedNudges = [];

            if (nudges && nudges.length > 0) {
                const nudgeIds = nudges.map(n => n.nudge_id);

                // BATCH 1: Get ALL impression stats in ONE aggregation query
                const [impressionStats, lastSession] = await Promise.all([
                    EventLog.aggregate([
                        {
                            $match: {
                                organization_id: orgId,
                                user_id: userId,
                                nudge_id: { $in: nudgeIds },
                                event_type: { $in: ['impression', 'campaign_impression', 'NINJA_EXPERIENCE_OPEN'] }
                            }
                        },
                        {
                            $facet: {
                                // Total impressions per nudge
                                total: [
                                    { $group: { _id: '$nudge_id', count: { $sum: 1 } } }
                                ],
                                // Daily impressions per nudge
                                daily: [
                                    {
                                        $match: {
                                            timestamp: { $gte: new Date(Date.now() - 24 * 60 * 60 * 1000) }
                                        }
                                    },
                                    { $group: { _id: '$nudge_id', count: { $sum: 1 } } }
                                ],
                                // Weekly impressions per nudge
                                weekly: [
                                    {
                                        $match: {
                                            timestamp: { $gte: new Date(Date.now() - 7 * 24 * 60 * 60 * 1000) }
                                        }
                                    },
                                    { $group: { _id: '$nudge_id', count: { $sum: 1 } } }
                                ]
                            }
                        }
                    ]),

                    // BATCH 2: Get session start (ONE query for all campaigns)
                    EventLog.findOne({
                        organization_id: orgId,
                        user_id: userId,
                        event_type: 'session_start'
                    }).sort({ timestamp: -1 }).lean()
                ]);

                // Convert stats to maps for O(1) lookup
                const totalMap = new Map((impressionStats[0]?.total || []).map(s => [s._id, s.count]));
                const dailyMap = new Map((impressionStats[0]?.daily || []).map(s => [s._id, s.count]));
                const weeklyMap = new Map((impressionStats[0]?.weekly || []).map(s => [s._id, s.count]));
                const sessionStartTime = lastSession?.timestamp || new Date(Date.now() - 30 * 60 * 1000);

                // BATCH 3: Get session impressions for all nudges in ONE query
                const sessionImpressions = await EventLog.aggregate([
                    {
                        $match: {
                            organization_id: orgId,
                            user_id: userId,
                            nudge_id: { $in: nudgeIds },
                            event_type: { $in: ['impression', 'campaign_impression', 'NINJA_EXPERIENCE_OPEN'] },
                            timestamp: { $gte: sessionStartTime }
                        }
                    },
                    {
                        $group: {
                            _id: '$nudge_id',
                            count: { $sum: 1 }
                        }
                    }
                ]);

                const sessionMap = new Map(sessionImpressions.map(s => [s._id, s.count]));

                // NOW iterate through nudges with pre-fetched data (NO MORE QUERIES!)
                for (const nudge of nudges) {
                    const totalImpressions = totalMap.get(nudge.nudge_id) || 0;
                    const dailyImpressions = dailyMap.get(nudge.nudge_id) || 0;
                    const weeklyImpressions = weeklyMap.get(nudge.nudge_id) || 0;
                    const sessImpressions = sessionMap.get(nudge.nudge_id) || 0;

                    // A. Platform Check
                    const targetPlatforms = (nudge.display_rules && nudge.display_rules.platforms) || nudge.target_audience;
                    if (targetPlatforms && targetPlatforms.length > 0) {
                        if (platform) {
                            const lowerPlatform = platform.toLowerCase();
                            const lowerTargets = targetPlatforms.map(p => p.toLowerCase());
                            // Flutter runs on both Android and iOS, so treat 'flutter' as matching either
                            const isFlutter = lowerPlatform === 'flutter';
                            const platformMatch = lowerTargets.includes(lowerPlatform) ||
                                (isFlutter && (lowerTargets.includes('android') || lowerTargets.includes('ios')));
                            if (!platformMatch) continue;
                        }
                    }

                    // A.0 Schedule Date Check - CRITICAL: Check if campaign is within its scheduled date range
                    if (nudge.schedule) {
                        const now = new Date();
                        const startDate = nudge.schedule.start_date ? new Date(nudge.schedule.start_date) : null;
                        const endDate = nudge.schedule.end_date ? new Date(nudge.schedule.end_date) : null;

                        // Skip if campaign hasn't started yet
                        if (startDate && now < startDate) continue;

                        // Skip if campaign has ended
                        if (endDate && now > endDate) continue;
                    }

                    // A.1 SDK Version Check
                    if (nudge.sdk_version_rule && nudge.sdk_version_rule.version) {
                        const sdkVersion = req.query.sdkVersion || req.headers['x-nudge-sdk-version'];
                        if (sdkVersion) {
                            const ruleVersion = nudge.sdk_version_rule.version;
                            if (nudge.sdk_version_rule.operator === 'equals' && sdkVersion !== ruleVersion) continue;
                            if (nudge.sdk_version_rule.operator === 'greater_than' && sdkVersion <= ruleVersion) continue;
                            if (nudge.sdk_version_rule.operator === 'less_than' && sdkVersion >= ruleVersion) continue;
                        }
                    }

                    // A.2 Segment Check
                    if (nudge.segments && nudge.segments.length > 0) {
                        const userSegments = userProfile?.segments || [];
                        const hasMatch = nudge.segments.some(seg => userSegments.includes(seg));
                        if (!hasMatch) continue;
                    }

                    // A.3 Frequency Capping (using pre-fetched stats!)
                    if (nudge.display_rules) {
                        const dr = nudge.display_rules;

                        // Legacy: frequency_cap (total lifetime limit)
                        if (dr.frequency_cap && totalImpressions >= dr.frequency_cap) continue;

                        // New: interactionLimit (total lifetime limit)
                        if (dr.interactionLimit && dr.interactionLimit.type === 'limited') {
                            const limitValue = dr.interactionLimit.value || 1;
                            if (totalImpressions >= limitValue) continue;
                        }

                        // New: frequency (time-based limits) - using pre-fetched stats
                        if (dr.frequency && dr.frequency.type !== 'every_time') {
                            const freqType = dr.frequency.type;
                            const freqValue = dr.frequency.value || 1;

                            if (freqType === 'daily' && dailyImpressions >= freqValue) continue;
                            if (freqType === 'weekly' && weeklyImpressions >= freqValue) continue;
                            // Monthly would need similar pre-fetching
                        }
                    }

                    // A.4 Session Capping (using pre-fetched session stats!)
                    if (nudge.display_rules) {
                        const dr = nudge.display_rules;

                        // Legacy: session_cap
                        if (dr.session_cap && sessImpressions >= dr.session_cap) continue;

                        // New: sessionLimit
                        if (dr.sessionLimit && dr.sessionLimit.enabled) {
                            const sessionLimitValue = dr.sessionLimit.value || 1;
                            if (sessImpressions >= sessionLimitValue) continue;
                        }

                        // New: frequency.type = once_per_session
                        if (dr.frequency && dr.frequency.type === 'once_per_session') {
                            if (sessImpressions >= 1) continue;
                        }
                    }

                    // B. Complex Targeting Rules
                    if (nudge.targeting && nudge.targeting.length > 0) {
                        let allRulesPassed = true;

                        for (const rule of nudge.targeting) {
                            if (rule.type === 'group') continue;

                            if (rule.type === 'user_property') {
                                const userValue = userProperties[rule.property || rule.field];
                                const targetValue = rule.value;
                                let rulePassed = false;

                                switch (rule.operator) {
                                    case 'equals': rulePassed = userValue == targetValue; break;
                                    case 'not_equals': rulePassed = userValue != targetValue; break;
                                    case 'greater_than': rulePassed = Number(userValue) > Number(targetValue); break;
                                    case 'greater_than_or_equal': rulePassed = Number(userValue) >= Number(targetValue); break;
                                    case 'less_than': rulePassed = Number(userValue) < Number(targetValue); break;
                                    case 'less_than_or_equal': rulePassed = Number(userValue) <= Number(targetValue); break;
                                    case 'contains': rulePassed = userValue != null && String(userValue).includes(String(targetValue)); break;
                                    case 'not_contains': rulePassed = userValue == null || !String(userValue).includes(String(targetValue)); break;
                                    case 'set': rulePassed = userValue !== undefined && userValue !== null; break;
                                    case 'not_set': rulePassed = userValue === undefined || userValue === null; break;
                                    default: rulePassed = true;
                                }
                                if (!rulePassed) {
                                    allRulesPassed = false;
                                    break;
                                }
                            }

                            // Handle Event Rules (these still need individual queries, but fewer nudges reach here)
                            if (rule.type === 'event') {
                                const eventName = rule.event || rule.property || rule.field;
                                const eventQuery = {
                                    organization_id: orgId,
                                    user_id: userId,
                                    event_type: eventName
                                };

                                if (rule.timeWindow && rule.timeWindow > 0) {
                                    eventQuery.timestamp = { $gte: new Date(Date.now() - rule.timeWindow * 1000) };
                                }

                                const count = await EventLog.countDocuments(eventQuery);
                                const targetValue = Number(rule.count || rule.value || 1);
                                let rulePassed = false;

                                const operator = rule.countOperator || rule.operator || 'greater_than_or_equal';

                                switch (operator) {
                                    case 'equals': rulePassed = count == targetValue; break;
                                    case 'not_equals': rulePassed = count !== targetValue; break;
                                    case 'greater_than': rulePassed = count > targetValue; break;
                                    case 'greater_than_or_equal': rulePassed = count >= targetValue; break;
                                    case 'less_than': rulePassed = count < targetValue; break;
                                    case 'less_than_or_equal': rulePassed = count <= targetValue; break;
                                    default: rulePassed = count >= targetValue;
                                }

                                if (!rulePassed) {
                                    allRulesPassed = false;
                                    break;
                                }
                            }
                        }

                        if (!allRulesPassed) continue;
                    }

                    matchedNudges.push(nudge);
                }
            }

            // ========== END OPTIMIZED QUERY BATCHING ==========

            // ========== END OPTIMIZED QUERY BATCHING ==========
            
            // Extract Gamification Progress from user profile to sync across devices (Phase 2)
            const challengeProgress = {};
            if (userProperties) {
                for (const key in userProperties) {
                    if (key.startsWith('_challenge_')) {
                        challengeProgress[key] = userProperties[key];
                    }
                }
            }

            // 4. Sanitize Gamification Configurations (Zero-Trust Anti-Cheat)
            const sanitizedNudges = matchedNudges.map(nudge => {
                // We create a deep clone so we don't accidentally mutate MongoDB return references
                const cleanNudge = JSON.parse(JSON.stringify(nudge));
                
                // Sanitize Spin The Wheel properties (check all possible storage locations)
                const stwSections = cleanNudge.config?.spinTheWheelConfig?.sections 
                    || cleanNudge.spinTheWheelConfig?.sections 
                    || cleanNudge.config?.sections;
                
                if (Array.isArray(stwSections)) {
                    const safeSections = stwSections.map(sec => {
                        const { weight, quantity, rewardId, audienceId, ...safeProps } = sec;
                        return safeProps;
                    });
                    // Write sanitized sections back to wherever they came from
                    if (cleanNudge.config?.spinTheWheelConfig?.sections) {
                        cleanNudge.config.spinTheWheelConfig.sections = safeSections;
                    }
                    if (cleanNudge.spinTheWheelConfig?.sections) {
                        cleanNudge.spinTheWheelConfig.sections = safeSections;
                    }
                    if (cleanNudge.config?.sections) {
                        cleanNudge.config.sections = safeSections;
                    }
                    // Also strip winningCriteria & sectionWinLimit from SDK payload
                    if (cleanNudge.config?.spinTheWheelConfig) {
                        delete cleanNudge.config.spinTheWheelConfig.winningCriteria;
                        delete cleanNudge.config.spinTheWheelConfig.sectionWinLimit;
                    }
                    if (cleanNudge.config?.winningCriteria) {
                        delete cleanNudge.config.winningCriteria;
                    }
                }
                
                // Sanitize Scratch Foil properties (if applicable)
                if (cleanNudge.scratchFoilConfig) {
                     const { quantity, rewardId, ...safeProps } = cleanNudge.scratchFoilConfig;
                     cleanNudge.scratchFoilConfig = safeProps;
                }
                
                return cleanNudge;
            });

            // 5. Return Response (Return ALL matches, not just one)
            return res.json({ data: sanitizedNudges, challengeProgress });

        } catch (error) {
            console.error('Critical Error in fetchNudge:', error);
            // Silent Fail Policy: Always return 200 OK with null data
            return res.json({ data: null });
        }
    }

    async trackEvent(req, res) {
        try {
            // Support SDK aliases
            let { action, metadata, userId, user_id, event, properties } = req.body;
            const orgId = req.orgId;

            // Normalize inputs
            action = action || event;
            metadata = metadata || properties || {};
            userId = userId || user_id;

            if (!action) {
                return res.status(400).json({ error: 'Action/Event is required' });
            }

            // Require Models (Safe Scope)
            const Event = require('../models/Event');
            const Property = require('../models/Property');
            const EndUser = require('../models/EndUser');
            const Nudge = require('../models/Nudge');
            const EventLog = require('../models/EventLog');
            const mongoose = require('mongoose');

            // Helper: Safe Regex Escape
            const escapeRegExp = (string) => string.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
            const safeAction = escapeRegExp(action);

            // Regex for flexible matching (spaces or underscores)
            // Example: "Checkout Step Viewed" matches "checkout_step_viewed"
            const fuzzyActionRegex = new RegExp(`^${safeAction.replace(/_/g, '[\\s_]')}$`, 'i');

            // A. Log Event
            const nudgeIdForLog = metadata?.nudgeId || metadata?.campaignId || null;
            await EventLog.create({
                event_type: action,
                metadata,
                user_id: userId,
                organization_id: orgId,
                nudge_id: nudgeIdForLog
            });

            // B. Auto-Discovery
            try {
                let eventDoc = await Event.findOne({
                    name: { $regex: new RegExp(`^${safeAction}$`, 'i') },
                    organization_id: orgId
                });

                if (!eventDoc) {
                    eventDoc = await Event.create({
                        name: action,
                        displayName: action.replace(/_/g, ' ').replace(/\b\w/g, l => l.toUpperCase()),
                        source: 'sdk_autodetected',
                        category: 'General',
                        validationLevel: 'none',
                        organization_id: orgId,
                        properties: []
                    });
                }

                if (metadata && Object.keys(metadata).length > 0) {
                    const existingProps = await Property.find({ organization_id: orgId }).select('name _id').lean();
                    const existingPropMap = new Map(existingProps.map(p => [p.name, p._id.toString()]));
                    const newPropIds = [];

                    for (const [key, value] of Object.entries(metadata)) {
                        let propId = existingPropMap.get(key);
                        if (!propId) {
                            let type = 'string';
                            if (typeof value === 'number') type = 'number';
                            if (typeof value === 'boolean') type = 'boolean';

                            const newProp = await Property.create({
                                name: key,
                                displayName: key.charAt(0).toUpperCase() + key.slice(1).replace(/_/g, ' '),
                                type,
                                organization_id: orgId
                            });
                            propId = newProp._id.toString();
                            existingPropMap.set(key, propId);
                        }
                        if (propId) newPropIds.push(propId);
                    }

                    if (newPropIds.length > 0) {
                        await Event.updateOne(
                            { _id: eventDoc._id },
                            { $addToSet: { properties: { $each: newPropIds } } }
                        );
                    }
                }
            } catch (discoveryError) {
                console.warn('Auto-Discovery Non-Fatal Error:', discoveryError.message);
            }

            // C. Update Stats
            if (nudgeIdForLog) {
            // Basic mapping to top-level stats (optional but helpful for fast queries)
            const lowerAction = action.toLowerCase();
            const update = {};
            if (['impression', 'view', 'shown', 'dismiss', 'close'].some(k => lowerAction.includes(k))) update['stats.impressions'] = 1;
            else if (['click', 'tap'].some(k => lowerAction.includes(k))) update['stats.clicks'] = 1;
            else if (['conversion', 'submit'].some(k => lowerAction.includes(k))) update['stats.conversions'] = 1;

                if (Object.keys(update).length > 0) {
                    const query = { organization_id: orgId };
                    if (mongoose.Types.ObjectId.isValid(nudgeIdForLog)) {
                        query.$or = [{ _id: nudgeIdForLog }, { nudge_id: nudgeIdForLog }];
                    } else {
                        query.nudge_id = nudgeIdForLog;
                    }
                    try { await Nudge.updateOne(query, { $inc: update }); } catch (e) { }
                }
            }

            // D. Real-time Matching
            console.log(`[Query] Searching Nudges for Org: ${orgId}, Action: ${action}`);

            const potentialNudges = await Nudge.find({
                organization_id: orgId,
                status: 'active',
                trigger_event: { $regex: fuzzyActionRegex }
            }).lean();

            const matchedNudges = [];

            if (potentialNudges.length > 0) {
                const userProfile = await EndUser.findOne({ organization_id: orgId, user_id: userId }).lean();
                const userProperties = userProfile ? { ...userProfile.properties, ...userProfile } : {};

                for (const nudge of potentialNudges) {
                    try {
                        // 1. Schedule
                        if (nudge.schedule) {
                            const now = new Date();
                            if (nudge.schedule.start_date && now < new Date(nudge.schedule.start_date)) continue;
                            if (nudge.schedule.end_date && now > new Date(nudge.schedule.end_date)) continue;
                        }

                        // 2. Event Count Rule
                        if (nudge.trigger_rules && nudge.trigger_rules.event_count) {
                            const count = await EventLog.countDocuments({
                                organization_id: orgId,
                                user_id: userId,
                                event_type: { $regex: fuzzyActionRegex }
                            });

                            const targetVal = Number(nudge.trigger_rules.value || nudge.trigger_rules.event_count);
                            const op = nudge.trigger_rules.operator || 'greater_than_or_equal';
                            let passed = false;

                            switch (op) {
                                case 'equals': passed = count === targetVal; break;
                                case 'not_equals': passed = count !== targetVal; break;
                                case 'greater_than': passed = count > targetVal; break;
                                case 'greater_than_or_equal': passed = count >= targetVal; break;
                                case 'less_than': passed = count < targetVal; break;
                                case 'less_than_or_equal': passed = count <= targetVal; break;
                                default: passed = count >= targetVal;
                            }
                            if (!passed) continue;
                        }

                        // 3. Segments
                        if (nudge.segments && nudge.segments.length > 0) {
                            const userSegments = userProfile?.segments || [];
                            if (!nudge.segments.some(seg => userSegments.includes(seg))) continue;
                        }

                        // 4. Frequency Cap
                        if (nudge.display_rules && nudge.display_rules.frequency_cap) {
                            const impressions = await EventLog.countDocuments({
                                organization_id: orgId,
                                user_id: userId,
                                nudge_id: nudge.nudge_id,
                                event_type: { $in: ['impression', 'campaign_impression'] }
                            });
                            if (impressions >= nudge.display_rules.frequency_cap) continue;
                        }

                        // 5. Targeting Rules
                        if (nudge.targeting && nudge.targeting.length > 0) {
                            let allRulesPassed = true;
                            for (const rule of nudge.targeting) {
                                if (rule.type === 'group') continue; // Simplified

                                if (rule.type === 'user_property') {
                                    const userValue = userProperties[rule.property || rule.field];
                                    const targetValue = rule.value;
                                    let rulePassed = false;

                                    // Handle numeric comparison specifically
                                    if (['greater_than', 'less_than', 'greater_than_or_equal', 'less_than_or_equal'].includes(rule.operator)) {
                                        const uv = Number(userValue);
                                        const tv = Number(targetValue);
                                        if (!isNaN(uv) && !isNaN(tv)) {
                                            if (rule.operator === 'greater_than') rulePassed = uv > tv;
                                            if (rule.operator === 'less_than') rulePassed = uv < tv;
                                            if (rule.operator === 'greater_than_or_equal') rulePassed = uv >= tv;
                                            if (rule.operator === 'less_than_or_equal') rulePassed = uv <= tv;
                                        }
                                    } else {
                                        // String/Equality
                                        if (rule.operator === 'equals') rulePassed = userValue == targetValue;
                                        else if (rule.operator === 'not_equals') rulePassed = userValue != targetValue;
                                        else if (rule.operator === 'contains') rulePassed = String(userValue).includes(String(targetValue));
                                        else if (rule.operator === 'not_contains') rulePassed = !String(userValue).includes(String(targetValue));
                                        else if (rule.operator === 'set') rulePassed = userValue != null;
                                        else if (rule.operator === 'not_set') rulePassed = userValue == null;
                                        else rulePassed = userValue == targetValue;
                                    }
                                    if (!rulePassed) { allRulesPassed = false; break; }
                                }
                            }
                            if (!allRulesPassed) continue;
                        }

                        matchedNudges.push(nudge);
                    } catch (loopError) {
                        console.error(`Skipping Nudge ${nudge._id} due to error:`, loopError);
                    }
                }
            }

            // 🛡️ SECURITY FIX: Apply same Zero-Trust sanitization as fetchNudge (lines 334-376)
            // Without this, rewardId, weight, quantity, and audienceId leak to the SDK client
            const sanitizedMatched = matchedNudges.map(nudge => {
                const cleanNudge = JSON.parse(JSON.stringify(nudge));
                
                // Sanitize Spin The Wheel sections
                const stwSections = cleanNudge.config?.spinTheWheelConfig?.sections
                    || cleanNudge.spinTheWheelConfig?.sections
                    || cleanNudge.config?.sections;
                
                if (Array.isArray(stwSections)) {
                    const safeSections = stwSections.map(sec => {
                        const { weight, quantity, rewardId, audienceId, ...safeProps } = sec;
                        return safeProps;
                    });
                    if (cleanNudge.config?.spinTheWheelConfig?.sections) {
                        cleanNudge.config.spinTheWheelConfig.sections = safeSections;
                    }
                    if (cleanNudge.spinTheWheelConfig?.sections) {
                        cleanNudge.spinTheWheelConfig.sections = safeSections;
                    }
                    if (cleanNudge.config?.sections) {
                        cleanNudge.config.sections = safeSections;
                    }
                    if (cleanNudge.config?.spinTheWheelConfig) {
                        delete cleanNudge.config.spinTheWheelConfig.winningCriteria;
                        delete cleanNudge.config.spinTheWheelConfig.sectionWinLimit;
                    }
                    if (cleanNudge.config?.winningCriteria) {
                        delete cleanNudge.config.winningCriteria;
                    }
                }
                
                // Sanitize Scratch Foil properties
                if (cleanNudge.scratchFoilConfig) {
                    const { quantity, rewardId, ...safeProps } = cleanNudge.scratchFoilConfig;
                    cleanNudge.scratchFoilConfig = safeProps;
                }
                
                return cleanNudge;
            });

            res.json({
                success: true,
                matched: sanitizedMatched
            });

        } catch (error) {
            console.error('Track Event Critical Error:', error);
            res.status(500).json({ error: error.message });
        }
    }

    async identifyUser(req, res) {
        try {
            const { userId, traits, referralCode } = req.body;
            const orgId = req.orgId;

            if (!userId) {
                return res.status(400).json({ error: 'userId is required' });
            }

            // 1. Auto-Discover Properties
            if (traits) {
                const Property = require('../models/Property');
                const existingProps = await Property.find({ organization_id: orgId }).select('name').lean();
                const existingPropNames = new Set(existingProps.map(p => p.name));

                const newProps = [];
                for (const [key, value] of Object.entries(traits)) {
                    if (!existingPropNames.has(key)) {
                        // Infer type
                        let type = 'string';
                        if (typeof value === 'number') type = 'number';
                        if (typeof value === 'boolean') type = 'boolean';

                        newProps.push({
                            name: key,
                            displayName: key.charAt(0).toUpperCase() + key.slice(1).replace(/_/g, ' '),
                            type,
                            organization_id: orgId,
                            description: 'Auto-discovered from SDK'
                        });
                        existingPropNames.add(key); // Prevent duplicates in this loop
                    }
                }

                if (newProps.length > 0) {
                    await Property.insertMany(newProps);
                    console.log(`[Auto-Discovery] Added ${newProps.length} new properties for Org ${orgId}`);
                }
            }

            // 2. Check for Lead Conversion
            const Lead = require('../models/Lead');
            const lead = await Lead.findOne({ organization_id: orgId, externalId: userId, status: 'pending' });
            if (lead) {
                lead.status = 'converted';
                lead.convertedAt = new Date();
                await lead.save();
                console.log(`[Lead Conversion] Lead ${userId} converted! Referred by ${lead.referredBy}`);
            }

            // 3. Upsert User
            const EndUser = require('../models/EndUser');
            const updateData = {
                name: traits?.name,
                email: traits?.email,
                phone: traits?.phone,
                properties: traits || {},
                last_seen: new Date(),
                device: traits?.device

exports.uploadPage = async (req, res) => {
    try {
        // 1. Validate Token
        const authHeader = req.headers.authorization;
        if (!authHeader) return res.status(401).json({ message: 'Missing token' });

        const token = authHeader.split(' ')[1];
        let decoded;
        try {
            decoded = jwt.verify(token, JWT_SECRET);
        } catch (err) {
            return res.status(403).json({ message: 'Invalid or expired token' });
        }

        if (decoded.type !== 'page_capture_session') {
            return res.status(403).json({ message: 'Invalid token type' });
        }

        const organization_id = decoded.organization_id;

        // 2. Validate File
        if (!req.file) {
            return res.status(400).json({ message: 'Screenshot image required' });
        }

        // 3. Optimize & Upload to Cloudinary
        // Use Sharp to resize/compress first, then proper stream upload
        const optimizedBuffer = await sharp(req.file.buffer)
            .resize(1080, null, { withoutEnlargement: true })
            .jpeg({ quality: 80 })
            .toBuffer();

        // Helper to upload stream to Cloudinary
        const uploadFromBuffer = (buffer) => {
            return new Promise((resolve, reject) => {
                const stream = cloudinary.uploader.upload_stream(
                    {
                        folder: 'embeddedcraft/pages', // Organized folder
                        resource_type: 'image'
                    },
                    (error, result) => {
                        if (result) {
                            resolve(result);
                        } else {
                            reject(error);
                        }
                    }
                );
                streamifier.createReadStream(buffer).pipe(stream);
            });
        };

        const result = await uploadFromBuffer(optimizedBuffer);
        const imageUrl = result.secure_url; // Verified Public URL

        // 4. Parse Metadata
        const {
            name,
            pageTag,
            elements,
            deviceMetadata
        } = req.body;

        let parsedElements = [];
        let parsedMetadata = {};

        try {
            parsedElements = typeof elements === 'string' ? JSON.parse(elements) : elements;
            parsedMetadata = typeof deviceMetadata === 'string' ? JSON.parse(deviceMetadata) : deviceMetadata;
        } catch (e) {
            return res.status(400).json({ message: 'Invalid JSON metadata' });
        }

        // 5. Save Page
        const newPage = new Page({
            organization_id,
            name: name || pageTag,
            pageTag,
            imageUrl: imageUrl, // Save full Cloudinary URL
            elements: parsedElements,
            deviceMetadata: parsedMetadata,
            sessionToken: token
        });

        await newPage.save();

        res.status(201).json({ message: 'Page uploaded successfully', pageId: newPage._id });

    } catch (error) {
        console.error('Upload Error:', error);
        res.status(500).json({ message: 'Upload failed: ' + error.message });
    }
};

// Multer: Memory storage (buffer) for Cloudinary streaming
// No local disk writes — files go directly to Cloudinary
const upload = multer({
    storage: multer.memoryStorage(),
    limits: { fileSize: 10 * 1024 * 1024 }, // 10MB
    fileFilter: (req, file, cb) => {
        const allowed = [
            'image/jpeg', 'image/png', 'image/gif', 'image/webp', 'image/svg+xml', 'image/avif',
            'video/mp4', 'video/webm',
            'application/pdf', 'application/json',
            'text/plain', 'text/csv'
        ];
        if (allowed.includes(file.mimetype) || file.mimetype.startsWith('image/')) {
            cb(null, true);
        } else {
            cb(new Error('Unsupported file type: ' + file.mimetype), false);
        }
    }
});
